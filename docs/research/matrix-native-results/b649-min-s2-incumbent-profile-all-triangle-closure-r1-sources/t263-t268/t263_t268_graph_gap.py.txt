def popcount(x: int) -> int:
    return x.bit_count()


def side(v: int) -> str:
    return "H" if v < N_H else "L"


# --------------------------------------------------------------------------- graph kernel


def build(h_missing, hl_edges, l_missing) -> list[int]:
    """Adjacency bitmasks: H = K12 minus h_missing, L = K8 minus l_missing, plus hl_edges."""
    adj = [0] * N

    def add(u: int, v: int) -> None:
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    hm = {frozenset(e) for e in h_missing}
    lm = {frozenset(e) for e in l_missing}
    for u, v in itertools.combinations(H, 2):
        if frozenset((u, v)) not in hm:
            add(u, v)
    for u, v in itertools.combinations(L, 2):
        if frozenset((u, v)) not in lm:
            add(u, v)
    for u, v in hl_edges:
        add(u, v)
    return adj


def edges_of(adj: list[int]) -> list[tuple[int, int]]:
    return [(u, v) for u in range(N) for v in range(u + 1, N) if adj[u] >> v & 1]


def degrees(adj: list[int]) -> list[int]:
    return [popcount(a) for a in adj]


def triangles(adj: list[int]) -> int:
    return sum(popcount(adj[u] & adj[v]) for u, v in edges_of(adj)) // 3


def wedges(adj: list[int]) -> int:
    return sum(comb(d, 2) for d in degrees(adj))


def open_wedges_direct(adj: list[int]) -> int:
    """Count open wedges centre-by-centre: neighbour pairs of v that are non-adjacent."""
    total = 0
    for v in range(N):
        nb = adj[v]
        closed = sum(popcount(adj[u] & nb) for u in range(N) if nb >> u & 1)
        total += comb(popcount(nb), 2) - closed // 2
    return total


def open_wedge_classes(adj: list[int]) -> Counter[str]:
    """Open wedges keyed 'centre side:end sides'; 'H:HL' = centre in H, one end in H, one in L."""
    cls: Counter[str] = Counter()
    for v in range(N):
        nb = [u for u in range(N) if adj[v] >> u & 1]
        for a, b in itertools.combinations(nb, 2):
            if not adj[a] >> b & 1:
                cls[side(v) + ":" + "".join(sorted(side(a) + side(b)))] += 1
    return cls


def arm_count(adj: list[int], u: int, v: int) -> int:
    """Open wedges having edge uv as an arm: deg u + deg v - 2 - 2 |N(u) & N(v)|."""
    return popcount(adj[u]) + popcount(adj[v]) - 2 - 2 * popcount(adj[u] & adj[v])


def h_internal_open_wedges(adj: list[int]) -> int:
    """Open wedges whose centre and both ends all lie in H, by brute force."""
    n = 0
    for centre in H:
        for a, b in itertools.combinations(H, 2):
            if a != centre and b != centre and adj[centre] >> a & 1 and adj[centre] >> b & 1:
                if not adj[a] >> b & 1:
                    n += 1
    return n


def hl_edge_count(adj: list[int]) -> int:
    return sum(popcount(adj[u] >> N_H) for u in H)


def f_degrees(f_edges) -> list[int]:
    d = [0] * N_H
    for a, b in f_edges:
        d[a] += 1
        d[b] += 1
    return d


# --------------------------------------------------------------------------- the three lemma bounds


def lemma_a_edge(dx: int, dz: int) -> int:
    """Common H-neighbours of a missing H pair xz: at least 12 - d_F(x) - d_F(z)."""
    return 12 - dx - dz


def lemma_b_edge(d: int) -> int:
    """H-centred mixed open wedges on an H-L edge ux: at least max(0, 5 - d_F(u))."""
    return max(0, 5 - d)


def lemma_c_edge(d: int, s: int) -> int:
    """Support refinement: N(x) & H lies in S = non-isolated F vertices, |S| = s."""
    return max(0, 11 - d - min(6, s - 1 - d))


def case_bound(c: int) -> dict[str, Any]:
    """Lower bound on O for e(H, L) = c > 0, as in proof step S9."""
    m = c // 2
    h_int = m * (11 - m)  # Lemma A with d_x + d_z <= m + 1
    if c == 2:
        mixed, rule = 2 * lemma_c_edge(1, 2), "LEMMA_A + LEMMA_C(s=2): 2 H-L edges x 10"
    elif c in (4, 6):
        mixed, rule = c * lemma_b_edge(m), f"LEMMA_A + LEMMA_B: {c} H-L edges x (5-{m})"
    else:
        mixed, rule = 0, "LEMMA_A alone"
    return {"M": m, "H_INTERNAL_LB": h_int, "MIXED_LB": mixed, "O_LOWER_BOUND": h_int + mixed, "RULE": rule}

# --------------------------------------------------------------------------- phase 1: arithmetic + parameter lemmas


def arm_lemma_min(bound: int) -> dict[str, Any]:
    """Exhaustive parameter check of the H-L arm lower bound; fails when bound is too high."""
    worst = min(
        (H_DEG - 1 - k) + (dy - 1 - k) for dy in L_DEGS for k in range(dy)  # k <= deg y - 1
    )
    return {"MIN_TERM": worst, "HOLDS": worst >= bound}


def lemma_b_params(offset: int) -> bool:
    """Every (d_F(u), deg x, overlap k) gives (11 - d) - k >= lemma_b_edge(d) + offset."""
    for d in range(1, N_L + 1):  # u has an L-neighbour, and deg_L(u) = d_F(u) <= 8
        for dx in L_DEGS:
            for k in range(min(dx - 1, 11 - d) + 1):  # k = |N_H(u) & N(x)| <= deg x - 1
                if (11 - d) - k < lemma_b_edge(d) + offset:
                    return False
    return True


def lemma_c_params(offset: int) -> bool:
    """Every (d, s, deg x, k) with k <= min(deg x - 1, s - 1 - d) meets lemma_c_edge(d, s) + offset."""
    for d in range(1, N_L + 1):
        for s in range(d + 1, N_H + 1):
            for dx in L_DEGS:
                for k in range(min(dx - 1, s - 1 - d, 11 - d) + 1):
                    if (11 - d) - k < lemma_c_edge(d, s) + offset:
                        return False
    return True


def lemma_a_params(offset: int) -> bool:
    """Missing pair xz: common H-neighbours = 10 - (d_x - 1) - (d_z - 1) + j, j = shared F-nbrs."""
    for m in range(1, 7):
        for dx in range(1, m + 1):
            for dz in range(1, m + 2 - dx):  # d_x + d_z <= m + 1
                for j in range(min(dx, dz)):
                    if 10 - (dx - 1) - (dz - 1) + j < lemma_a_edge(dx, dz) + offset:
                        return False
    return True


def per_cell(t: int, bounds: dict[int, dict[str, Any]]) -> dict[str, Any]:
    o = W_EXPECTED - 3 * t
    c_max = (2 * o) // arm_lemma_min(4)["MIN_TERM"]
    admissible = [c for c in range(c_max + 1) if c % 2 == 0]
    exclusion: dict[str, str] = {}
    closed = True
    for c in admissible:
        if c == 0:
            exclusion["0"] = f"FORCES_K12_UNION_K8_MINUS_E_T{T_INCUMBENT}"
            closed &= t != T_INCUMBENT
            continue
        lb = bounds[c]["O_LOWER_BOUND"] if c in bounds else 2 * c
        exclusion[str(c)] = f"O>={lb}>{o}" if lb > o else f"OPEN:O>={lb}<={o}"
        closed &= lb > o
    return {
        "O": o,
        "C_MAX_FROM_4C_LE_2O": c_max,
        "C_ADMISSIBLE_AFTER_PARITY": admissible,
        "EXCLUSION_BY_C": exclusion,
        "STATUS": "PROVED_EMPTY" if closed else "NOT_CLOSED_BY_THIS_ARGUMENT",
    }

# --------------------------------------------------------------------------- phase 2: F degree sequences


def partitions(total: int, max_part: int):
    if total == 0:
        yield ()
        return
    for p in range(min(total, max_part), 0, -1):
        for rest in partitions(total - p, p):
            yield (p, *rest)


def erdos_gallai(seq: tuple[int, ...]) -> bool:
    if sum(seq) % 2:
        return False
    for k in range(1, len(seq) + 1):
        if sum(seq[:k]) > k * (k - 1) + sum(min(d, k) for d in seq[k:]):
            return False
    return True


def havel_hakimi(seq: tuple[int, ...]) -> bool:
    rem = sorted(seq, reverse=True)
    while rem and rem[0] > 0:
        d, rem = rem[0], rem[1:]
        if d > len(rem):
            return False
        rem = [x - 1 for x in rem[:d]] + rem[d:]
        if min(rem[:d], default=0) < 0:
            return False
        rem.sort(reverse=True)
    return True


def phase2() -> dict[str, Any]:
    """All graphic degree sequences of F (m edges, <= 12 vertices): exact minima of each bound."""
    out: dict[str, Any] = {}
    for m in range(1, 7):

        for p in cand:
            assert erdos_gallai(p) == havel_hakimi(p), p
        seqs = [p for p in cand if erdos_gallai(p)]
        h_int = {p: 12 * m - sum(d * d for d in p) for p in seqs}
        mix_b = {p: sum(d * lemma_b_edge(d) for d in p) for p in seqs}
        mix_c = {p: sum(d * lemma_c_edge(d, len(p)) for d in p) for p in seqs}
        star = (m, *(1,) * m)
        assert star in h_int and h_int[star] == min(h_int.values()) == m * (11 - m)
        assert min(mix_b.values()) >= 2 * m * lemma_b_edge(m)
        b = case_bound(2 * m)
        mixed_rule = mix_c if m == 1 else mix_b  # c = 2 uses Lemma C, c >= 4 uses Lemma B
        assert min(h_int[p] + mixed_rule[p] for p in seqs) >= b["O_LOWER_BOUND"]
        if m == 1:
            assert min(h_int[p] + mix_b[p] for p in seqs) < TARGET  # Lemma B alone cannot do c = 2
        out[str(m)] = {
            "GRAPHIC_SEQUENCES": len(seqs),

# --------------------------------------------------------------------------- phase 3: labelled F, m <= 3


def h_internal_fast(f_edges) -> tuple[int, list[int]]:
    """Exact H-internal open wedges of K12 - F: sum over missing pairs of common H-neighbours."""
    hadj = [H_MASK ^ (1 << u) for u in H]
    for a, b in f_edges:
        hadj[a] &= ~(1 << b)
        hadj[b] &= ~(1 << a)
    return sum(popcount(hadj[a] & hadj[b]) for a, b in f_edges), [popcount(hadj[a] & hadj[b]) for a, b in f_edges]


def phase3() -> dict[str, Any]:
    out: dict[str, Any] = {}
    pairs = list(itertools.combinations(H, 2))
    for m in (1, 2, 3):
        n = 0
        minimum = None
        for f in itertools.combinations(pairs, m):
            d = f_degrees(f)
            total, per_edge = h_internal_fast(f)
            for (a, b), cn in zip(f, per_edge, strict=True):
                assert d[a] + d[b] <= m + 1
                assert cn >= lemma_a_edge(d[a], d[b])
            assert total >= m * (11 - m)
            if m <= 2:
                assert total == h_internal_open_wedges(build(f, [], []))  # brute-force cross-check
            minimum = total if minimum is None else min(minimum, total)
            n += 1
        assert n == comb(len(pairs), m) and minimum == m * (11 - m)
        out[str(m)] = {"LABELLED_F": n, "MIN_EXACT_H_INTERNAL": minimum, "BRUTE_FORCE_CROSS_CHECK": m <= 2}
    return out


# --------------------------------------------------------------------------- phase 4: concrete graphs


def hh_edges(seq: list[int], rng: random.Random) -> list[tuple[int, int]] | None:
    """Havel-Hakimi realisation with random tie-breaks; None when it gets stuck."""
    rem = list(seq)
    edges: list[tuple[int, int]] = []
    while True:
        order = sorted(range(len(rem)), key=lambda i: (-rem[i], rng.random()))
        v = order[0]
        if rem[v] == 0:
            return edges
        targets = order[1 : 1 + rem[v]]
        if len(targets) < rem[v] or any(rem[t] == 0 for t in targets):
            return None
        for t in targets:
            rem[t] -= 1
            edges.append((v, t))
        rem[v] = 0


def realize(f_edges, rng: random.Random, tries: int = 400) -> list[int] | None:
    """A degree-valid G with G[H] = K12 - F: H-L graph with deg_L(u) = d_F(u), L = K8 - R."""
    d = f_degrees(f_edges)
    c = 2 * len(f_edges)
    s = sum(1 for x in d if x)
    cap = [min(L_DEG_BY_INDEX[i], s) for i in range(N_L)]
    for _ in range(tries):
        cx = [0] * N_L
        for _ in range(c):
            open_slots = [i for i in range(N_L) if cx[i] < cap[i]]
            cx[rng.choice(open_slots)] += 1
        rem, hl, ok = cx[:], [], True
        for u in sorted(H, key=lambda u: (-d[u], rng.random())):
            if d[u]:
                pick = sorted(range(N_L), key=lambda i: (-rem[i], rng.random()))[: d[u]]
                if any(rem[i] == 0 for i in pick):
                    ok = False
                    break
                for i in pick:
                    rem[i] -= 1
                    hl.append((u, N_H + i))
        if not ok:
            continue
        r_edges = hh_edges([(7 - L_DEG_BY_INDEX[i]) + cx[i] for i in range(N_L)], rng)
        if r_edges is None:
            continue
        adj = build(f_edges, hl, [(N_H + i, N_H + j) for i, j in r_edges])
        assert sorted(degrees(adj)) == DEG_SORTED and hl_edge_count(adj) == c
        return adj
    return None


def check_graph(adj: list[int], stats: dict[int, dict[str, Any]]) -> None:
    """Every identity and lemma of the proof on one degree-valid graph."""
    degs = degrees(adj)
    assert sorted(degs) == DEG_SORTED
    t = triangles(adj)
    o = open_wedges_direct(adj)
    assert wedges(adj) == W_EXPECTED and o == W_EXPECTED - 3 * t  # O two independent ways
    cls = open_wedge_classes(adj)
    assert sum(cls.values()) == o  # the six classes partition the open wedges
    c = hl_edge_count(adj)
    f_edges = [(x, z) for x, z in itertools.combinations(H, 2) if not adj[x] >> z & 1]
    m = len(f_edges)
    assert c == 2 * m and c % 2 == 0
    d = [H_DEG - popcount(adj[u] & H_MASK) for u in H]
    assert d == f_degrees(f_edges)
    assert all(popcount(adj[u] >> N_H) == d[u] for u in H)  # deg_L(u) = d_F(u)
    s_mask = sum(1 << u for u in H if d[u])
    s = popcount(s_mask)
    assert s <= 2 * m
    hh = 0
    for x, z in f_edges:
        cn = popcount(adj[x] & adj[z] & H_MASK)
        assert d[x] + d[z] <= m + 1 and cn >= lemma_a_edge(d[x], d[z])
        hh += cn
    assert hh == cls["H:HH"]  # Lemma A wedges are exactly class H:HH, one end pair each
    if c in C_PROOF:
        assert hh >= m * (11 - m)
    hl_sum = 0
    for x in L:
        assert adj[x] & H_MASK & ~s_mask == 0  # N(x) & H lies in S
    for u in H:
        for x in L:
            if adj[u] >> x & 1:
                assert d[u] <= m and popcount(adj[x] & H_MASK) <= degs[x] <= 7
                mixed = popcount(adj[u] & H_MASK & ~adj[x])  # w in N_H(u), w not in N(x)
                assert mixed >= lemma_b_edge(d[u]) and mixed >= lemma_c_edge(d[u], s)
                hl_sum += mixed
    assert hl_sum == cls["H:HL"]  # one H-L arm per H-centred mixed wedge
    arm_sum = 0
    for u, v in edges_of(adj):
        a = arm_count(adj, u, v)
        assert a >= 0 and (side(u) == side(v) or a >= 4)
        arm_sum += a
    assert arm_sum == 2 * o and 4 * c <= 2 * o
    if c == 0:
        assert t == T_INCUMBENT and o == 6
    else:
        lb = case_bound(c)["O_LOWER_BOUND"] if c in C_PROOF else 2 * c
        assert o >= lb >= TARGET, (c, o, lb)
    assert t not in CELLS
    rec = stats.setdefault(c, {"GRAPHS": 0, "MIN_O": o, "MAX_T": t, "MIN_H_HH": hh, "MIN_H_HL": cls["H:HL"]})
    rec["GRAPHS"] += 1
    rec["MIN_O"], rec["MAX_T"] = min(rec["MIN_O"], o), max(rec["MAX_T"], t)
    rec["MIN_H_HH"], rec["MIN_H_HL"] = min(rec["MIN_H_HH"], hh), min(rec["MIN_H_HL"], cls["H:HL"])


def catalogue(m: int, rng: random.Random) -> list[tuple[str, list[tuple[int, int]]]]:
    pairs = list(itertools.combinations(H, 2))
    cat = [
        ("STAR", [(0, i) for i in range(1, m + 1)]),
        ("MATCHING", [(2 * i, 2 * i + 1) for i in range(m)]),
        ("PATH", [(i, i + 1) for i in range(m)]),
    ]
    if m >= 3:
        cat.append(("CYCLE", [(i, (i + 1) % m) for i in range(m)]))
    if m >= 4:
        cat.append(("TRIANGLE_PLUS_MATCHING", [(0, 1), (1, 2), (0, 2)] + [(3 + 2 * i, 4 + 2 * i) for i in range(m - 3)]))
        cat.append(("STAR3_PLUS_PATH", [(0, 1), (0, 2), (0, 3)] + [(3 + i, 4 + i) for i in range(m - 3)]))
    if m == 6:
        cat.append(("K4", list(itertools.combinations(range(4), 2))))
        cat.append(("TWO_TRIANGLES", [(0, 1), (1, 2), (0, 2), (3, 4), (4, 5), (3, 5)]))
    cat += [(f"RANDOM_{r}", rng.sample(pairs, m)) for r in range(RANDOM_F_PER_M)]
    return cat


def phase4() -> dict[str, Any]:
    rng = random.Random(SEED)
    stats: dict[int, dict[str, Any]] = {}
    # c = 0: every labelled choice of the missing L edge gives K12 + (K8 - e) with T = 270.
    for e in itertools.combinations(L, 2):
        adj = build([], [], [e])
        assert sorted(degrees(adj)) == DEG_SORTED
        assert triangles(adj) == comb(12, 3) + comb(8, 3) - 6 == 270
        check_graph(adj, stats)
    assert stats[0]["GRAPHS"] == 28
    realized: dict[str, dict[str, int]] = {}
    for m in range(1, 7):
        row: dict[str, Any] = {"F_TRIED": 0, "GRAPHS_CHECKED": 0, "UNREALIZED_F": []}
        for name, f in catalogue(m, rng):
            row["F_TRIED"] += 1
            got = 0
            for _ in range(REALIZATIONS_PER_F):
                adj = realize(f, rng)
                if adj is not None:
                    check_graph(adj, stats)
                    got += 1
            row["GRAPHS_CHECKED"] += got
            if not got:
                row["UNREALIZED_F"].append(name)
        assert row["GRAPHS_CHECKED"] > 0  # not vacuous
        assert "STAR" not in row["UNREALIZED_F"] and "MATCHING" not in row["UNREALIZED_F"]
        realized[str(m)] = row
    for c in C_PROOF:
        assert stats[c]["GRAPHS"] > 0 and stats[c]["MIN_O"] >= case_bound(c)["O_LOWER_BOUND"]
    walk = short_walks(stats)
    witness = arm4_witness(stats)
    return {
        "BY_C": {str(c): v for c, v in sorted(stats.items())},
        "REALIZED_CATALOGUE_BY_M": realized,
        "SHORT_WALKS": walk,
        "ARM4_TIGHTNESS_WITNESS": witness,
    }


def switch(adj: list[int], edges: list[tuple[int, int]], rng: random.Random) -> bool:
    """One degree-preserving double-edge switch ab, cd -> ad, cb; False when rejected."""
    i, j = rng.sample(range(len(edges)), 2)
    a, b = edges[i]
    c_, d = edges[j]
    if rng.random() < 0.5:
        c_, d = d, c_
    if len({a, b, c_, d}) < 4 or adj[a] >> d & 1 or adj[c_] >> b & 1:
        return False
    for x, y in ((a, b), (c_, d)):
        adj[x] &= ~(1 << y)
        adj[y] &= ~(1 << x)
    for x, y in ((a, d), (c_, b)):
        adj[x] |= 1 << y
        adj[y] |= 1 << x
    edges[i], edges[j] = (min(a, d), max(a, d)), (min(c_, b), max(c_, b))
    return True


def arm4_witness(stats: dict[int, dict[str, Any]], max_steps: int = 4000) -> dict[str, Any]:
    """A degree-valid graph with an H-L edge whose arm count is exactly 4: the S2 bound is
    attained by a real graph, not only by its parameters. Found by a seeded switch walk."""
    rng = random.Random(SEED + 2)
    adj = build([], [], [(N - 2, N - 1)])
    edges = edges_of(adj)
    for step in range(1, max_steps + 1):
        if not switch(adj, edges, rng):
            continue
        arms = [arm_count(adj, u, v) for u, v in edges_of(adj) if u < N_H <= v]
        if arms and min(arms) == 4:
            check_graph(adj, stats)
            t = triangles(adj)
            return {"SEED": SEED + 2, "STEP": step, "C": hl_edge_count(adj), "MIN_HL_ARM": 4, "T": t, "O": W_EXPECTED - 3 * t}
    raise AssertionError("no arm-4 witness found")


def short_walks(stats: dict[int, dict[str, Any]]) -> dict[str, Any]:
    """Degree-preserving switch walks restarted at K12 + (K8 - e); every visit with c <= 12 is
    fully checked, every other visit gets O >= 2c >= 28 and T not in the cells. Non-load-bearing."""
    rng = random.Random(SEED + 1)
    accepted = full = light = 0
    for _ in range(WALK_RESTARTS):
        adj = build([], [], [(N - 2, N - 1)])
        edges = edges_of(adj)
        for _ in range(WALK_STEPS):
            if not switch(adj, edges, rng):
                continue
            accepted += 1
            cc = hl_edge_count(adj)
            if cc <= 12:
                check_graph(adj, stats)
                full += 1
            else:
                t = triangles(adj)
                o = W_EXPECTED - 3 * t
                assert o == open_wedges_direct(adj) and o >= 2 * cc >= TARGET and t not in CELLS
                light += 1
    return {
        "SEED": SEED + 1,
        "RESTARTS": WALK_RESTARTS,
        "STEPS_PER_RESTART": WALK_STEPS,
        "ACCEPTED_SWITCHES": accepted,
        "FULL_CHECKS_C_LE_12": full,
        "LIGHT_CHECKS_C_GE_14": light,
        "CELL_VISITS": 0,
        "NON_LOAD_BEARING": True,
    }

def cpu_seconds() -> float:
    s = resource.getrusage(resource.RUSAGE_SELF)
    c = resource.getrusage(resource.RUSAGE_CHILDREN)
    return s.ru_utime + s.ru_stime + c.ru_utime + c.ru_stime


def predecessor_check() -> dict[str, str]:
    out = {}
    for key, (sha, path) in PREDECESSORS.items():
        p = Path(path)
        if p.exists():
            assert hashlib.sha256(p.read_bytes()).hexdigest() == sha, key
            out[f"{key}_RESULT_SHA256"] = sha
            out[f"{key}_READ_ONLY_HASH_CHECK"] = "MATCH"
        else:
            out[f"{key}_RESULT_SHA256"] = sha
            out[f"{key}_READ_ONLY_HASH_CHECK"] = "ABSENT"
    return out
